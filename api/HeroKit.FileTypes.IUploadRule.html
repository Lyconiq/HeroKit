Examples

sealed class NoEmptyPdfRule : IUploadRule
{
    public string? Validate(ReadOnlySpan<byte> header, string fileName, FileType? detectedType) =>
        detectedType == FileTypes.Pdf && header.Length < 100 ? "The PDF is too small." : null;
}
